% Wrap every lstlisting in a tcolorbox: one solid panel, so no
% white stripes between lines (which is what backgroundcolor gives).
% Tip badges. One icon + colour per category; unknown categories
% fall back to the last pair. See etc/code.lua.
\definecolor{tipink}{HTML}{1F4E9C}
\newcommand\tipfor[3]{%
  \expandafter\def\csname tip@i@#1\endcsname{#2}%
  \expandafter\def\csname tip@c@#1\endcsname{#3}}
\tipfor{installs}{\faBoxOpen}{B4620A}
\tipfor{doco}    {\faBookOpen}{0A6B4B}
\tipfor{edit}    {\faKeyboard}{6B3FA0}
\tipfor{lisp}    {\faCode}    {1F4E9C}
\tipfor{scripting}{\faTerminal}{9C1F4E}
% Colour alone carries the category -- quieter than an icon per
% tip. Flip \tipiconsfalse to \tipiconstrue to put the icons back.
\newif\iftipicons \tipiconsfalse
\newcommand\tip[2]{%
  \ifcsname tip@i@#1\endcsname
    \definecolor{tipink}{HTML}{\csname tip@c@#1\endcsname}%
    \iftipicons\textcolor{tipink}{\csname tip@i@#1\endcsname}\,\fi
  \else \iftipicons\textcolor{tipink}{\faLightbulb}\,\fi\fi
  \textcolor{tipink}{\textbf{TIP#2}}%
  \,\textcolor{tipink}{\footnotesize\textsc{(#1)}}}

% Two listing environments (see etc/code.lua).
% lispcode = real code: coloured, on a gray panel.
\lstnewenvironment{lispcode}{}{}
% egcode = examples quoted inside the prose: plain black, no panel,
% so they never read as "here is the source".
\lstnewenvironment{egcode}
  {\lstset{basicstyle=\ttfamily\scriptsize\color{black!85},
            keywordstyle=\ttfamily, commentstyle=\ttfamily,
            stringstyle=\ttfamily, xleftmargin=10pt,
            aboveskip=4pt, belowskip=3pt}}{}

\tcolorboxenvironment{lispcode}{
  colback=bg, colframe=rule, sharp corners,
  boxrule=0pt, leftrule=1.2pt,
  left=6pt, right=2pt, top=3pt, bottom=3pt,
  boxsep=0pt, breakable,
  before skip=5pt, after skip=4pt}

% Inline `code` in a proportional-ish mono, slightly smaller.
